# 3. conky v2.1 Technical Delta (`conky.wit` & A2UI)

## 3.1 Updated `conky.wit` Airlock Interface

```wit
package conky:kernel@2.1.0;

interface types {
  type crest = string;
  type anchor = string;
  type seal = string;

  record line {
    vault: string,
    unit: string,
    shift: float64,
  }

  record patch {
    target: anchor,
    delta: string,
    parity: list<line>,
    shadow: bool,          // Executes against in-memory libgit2 ODB
    settle: string,        // "eager" | "quiescent"
    purpose: anchor,
    base: seal,
  }

  record clone-spec {
    template: anchor,      // Certified source triad (e.g. "[[@realm-template]]")
    new-crest: crest,      // Target child crest (e.g. "@realm-southbank")
    attenuate: list<string>, // Subset of parent grants
    purpose: anchor,
  }

  variant fault {
    sprawl(u32),
    ward(string),
    guard-rejected(string),
    unproven(string),
    gate(anchor),
    imbalance,
    cycle(u8),
    taint,
  }
}

interface airlock {
  use types.{anchor, patch, clone-spec, seal, fault};

  peek: func(target: anchor) -> result<string, fault>;
  watch: func(target: anchor, when: string) -> result<u64, fault>;
  etch: func(req: patch) -> result<seal, fault>;
  clone: func(spec: clone-spec) -> result<seal, fault>; // 1-breath atomic triad spawn
}
```
